## 3. 核心算法实现

### 3.1 字母序列生成算法

#### 3.1.1 随机模式 (Random Mode)
```javascript
/**
 * 生成随机字母序列
 * @param {number} length - 序列长度
 * @param {boolean} caseSensitive - 是否区分大小写
 * @param {boolean} includePunctuation - 是否包含标点
 * @returns {string[]} 字母序列
 */
function generateRandomSequence(length, caseSensitive, includePunctuation) {
  const letters = 'abcdefghijklmnopqrstuvwxyz';
  const punctuation = '.,!?;:';
  const numbers = '0123456789';
  let charset = letters;

  // 根据难度调整字符集
  if (caseSensitive) charset += letters.toUpperCase();
  if (includePunctuation) charset += punctuation + numbers;

  const sequence = [];
  let lastChar = '';

  for (let i = 0; i < length; i++) {
    let char;
    let attempts = 0;
    const maxAttempts = 10;

    // 避免连续3个以上相同字符
    do {
      char = charset[Math.floor(Math.random() * charset.length)];
      attempts++;
    } while (
      attempts < maxAttempts &&
      (char === lastChar || char === lastChar.repeat(2))
    );

    sequence.push(char);
    lastChar = char;
  }

  return sequence;
}
```

#### 3.1.2 单词模式 (Word Mode)
```javascript
/**
 * 从单词列表生成字母序列
 * @param {string[]} wordList - 单词列表
 * @param {number} length - 序列长度
 * @returns {string[]} 字母序列
 */
function generateFromWords(wordList, length) {
  const sequence = [];
  let currentWord = '';

  while (sequence.length < length) {
    if (currentWord.length === 0) {
      currentWord = wordList[Math.floor(Math.random() * wordList.length)];
    }

    sequence.push(currentWord[0]);
    currentWord = currentWord.slice(1);

    // 在单词间插入空格（随机）
    if (Math.random() < 0.3 && currentWord.length > 0) {
      sequence.push(' ');
      currentWord = currentWord.slice(1);
    }
  }

  return sequence;
}
```

### 3.2 计分算法

#### 3.2.1 基础得分
```javascript
/**
 * 计算基础得分
 * @param {boolean} isCorrect - 是否正确
 * @param {number} combo - 当前连击数
 * @param {number} wpm - 当前WPM
 * @returns {number} 基础得分
 */
function calculateBaseScore(isCorrect, combo, wpm) {
  if (isCorrect) {
    let score = 10;

    // 连击奖励
    if (combo >= 10 && combo < 20) score += 20;
    else if (combo >= 20 && combo < 30) score += 50;
    else if (combo >= 30) score += 100;

    // 速度奖励
    if (wpm >= 20 && wpm <= 30) score *= 1.1;
    else if (wpm > 30 && wpm <= 40) score *= 1.2;
    else if (wpm > 40) score *= 1.3;

    return Math.round(score);
  } else {
    return -5;
  }
}
```

#### 3.2.2 最终得分计算
```javascript
/**
 * 计算最终得分
 * @param {number} baseScore - 基础得分
 * @param {number} accuracy - 准确率
 * @param {number} timeBonus - 时间奖励
 * @returns {number} 最终得分
 */
function calculateFinalScore(baseScore, accuracy, timeBonus) {
  let finalScore = baseScore;

  // 准确率惩罚
  if (accuracy < 0.8) {
    finalScore *= 0.7;
  }

  // 时间奖励（提前完成）
  finalScore += timeBonus;

  return Math.max(0, Math.round(finalScore));
}
```

### 3.3 WPM计算算法

```javascript
/**
 * 计算WPM (Words Per Minute)
 * @param {number} correctChars - 正确字符数
 * @param {number} elapsedTime - 已用时间（秒）
 * @returns {number} WPM
 */
function calculateWPM(correctChars, elapsedTime) {
  if (elapsedTime === 0) return 0;

  // 假设1个单词 = 5个字符
  const words = correctChars / 5;
  const minutes = elapsedTime / 60;

  return Math.round(words / minutes);
}
```

## 5. 事件流程设计

### 5.1 游戏初始化流程
```
开始
  ↓
加载配置 (PracticeConfig)
  ↓
生成字母序列 (CharacterGenerator)
  ↓
初始化游戏状态 (GameState)
  ↓
绑定键盘事件 (KeyboardHandler)
  ↓
等待用户按键
```

### 5.2 游戏运行流程
```
用户按键
  ↓
KeyboardHandler.handleKeyPress()
  ↓
验证键位有效性
  ↓
GameEngine.handleKeyPress()
  ↓
检查正确性
  ├─正确→ 更新得分、连击、WPM
  └─错误→ 扣分、连击归零、错误反馈
  ↓
检查序列是否完成
  ├─未完成→ 序列左移，补充新字符
  └─已完成→ 生成新序列
  ↓
更新时间剩余
  ↓
触发 onScoreChange 事件
  ↓
更新UI渲染
  ↓
等待下次按键
```

### 5.3 练习结束流程
```
倒计时结束 或 用户主动结束
  ↓
计算最终得分
  ↓
保存练习记录 (StorageManager)
  ↓
显示结果页面
  ↓
触发 onGameFinish 事件
  ↓
解除键盘绑定
  ↓
等待重新开始
```

## 7. 性能要求与优化

### 7.1 性能指标

| 指标类型 | 目标值 | 备注 |
|----------|--------|------|
| 页面加载时间 | < 2秒 | 首次加载 |
| 键盘响应延迟 | < 50ms | 从按键到视觉反馈 |
| 动画帧率 | 60 FPS | 流畅度 |
| 内存占用 | < 100MB | 运行时 |
| CPU占用 | < 10% | 空闲时 |
| 电池消耗 | 低 | 移动端友好 |

### 7.2 性能优化策略

#### 7.2.1 渲染优化
- 使用 `requestAnimationFrame` 进行动画
- 避免频繁DOM操作，批量更新
- 使用 CSS transform 而非改变 position
- 虚拟化长列表（如果序列很长）

#### 7.2.2 内存优化
- 及时清理事件监听器
- 避免在循环中创建对象
- 复用对象池（如音效对象）
- 及时清除定时器

#### 7.2.3 算法优化
- 预生成字母序列（一次性生成500个字符）
- 使用 Map 替代对象查找（提高键位映射性能）
- 缓存计算结果（WPM、准确率）

## 8. 错误处理机制

### 8.1 错误分类

| 错误类型 | 错误码 | 描述 | 处理方案 |
|----------|--------|------|----------|
| INVALID_CONFIG | 1001 | 配置无效 | 使用默认配置 |
| GENERATION_FAILED | 2001 | 序列生成失败 | 回退到随机模式 |
| STORAGE_ERROR | 3001 | 存储失败 | 使用内存存储，提示用户 |
| AUDIO_ERROR | 4001 | 音效加载失败 | 静默失败，继续游戏 |
| TIMER_ERROR | 5001 | 定时器异常 | 重新初始化定时器 |

### 8.2 错误处理代码示例
```javascript
// 全局错误处理
window.addEventListener('error', (e) => {
  console.error('Global Error:', e.error);
  // 记录错误日志
  logError(e.error);
  // 显示用户友好提示
  showUserMessage('出现了一些问题，但不影响继续练习。');
});

// Promise 错误处理
function safeAsync(fn) {
  return Promise.resolve()
    .then(fn)
    .catch((error) => {
      console.error('Async Error:', error);
      logError(error);
      return null;
    });
}

// 使用示例
safeAsync(() => generator.generate(config))
  .then((sequence) => {
    if (!sequence) throw new Error('GENERATION_FAILED');
    // 继续处理
  });
```

```javascript
// 工具函数库 (utils.js)
const Utils = {
  // 格式化时间 (MM:SS)
  formatTime(seconds) {
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  },

  // 防抖函数
  debounce(func, wait) {
    let timeout;
    return function executedFunction(...args) {
      const later = () => {
        clearTimeout(timeout);
        func(...args);
      };
      clearTimeout(timeout);
      timeout = setTimeout(later, wait);
    };
  },

  // 节流函数
  throttle(func, limit) {
    let inThrottle;
    return function() {
      const args = arguments;
      const context = this;
      if (!inThrottle) {
        func.apply(context, args);
        inThrottle = true;
        setTimeout(() => inThrottle = false, limit);
      }
    };
  },

  // 生成随机ID
  generateId() {
    return Date.now().toString(36) + Math.random().toString(36).substr(2);
  },

  // 深拷贝
  deepClone(obj) {
    return JSON.parse(JSON.stringify(obj));
  }
};
```

### 14.2 事件总线实现

```javascript
// 简单事件总线
class EventBus {
  constructor() {
    this.events = {};
  }

  on(eventName, callback) {
    if (!this.events[eventName]) {
      this.events[eventName] = [];
    }
    this.events[eventName].push(callback);
  }

  emit(eventName, data) {
    if (this.events[eventName]) {
      this.events[eventName].forEach(callback => callback(data));
    }
  }

  off(eventName, callback) {
    if (this.events[eventName]) {
      this.events[eventName] = this.events[eventName].filter(cb => cb !== callback);
    }
  }
}
```
